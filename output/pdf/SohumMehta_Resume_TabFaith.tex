%-------------------------- CUSTOM COMMANDS -----------------------------------
% A bulleted achievement line. Keep each one to a single printed line.
\newcommand{\resumeItem}[1]{%
  \item\small{#1 \vspace{-3pt}}
}

% One-line entry header:  Organization | Position ................... Dates
\newcommand{\resumeEntry}[3]{%
  \vspace{-2pt}\item
    \begin{tabular*}{0.97\textwidth}[t]{l@{\extracolsep{\fill}}r}
      \textbf{#1} $|$ \textit{\small #2} & \textit{\small #3} \\
    \end{tabular*}\vspace{-7pt}
}

% Two-line education header: School .. Location / Degree .. Dates
\newcommand{\resumeEdu}[4]{%
  \vspace{-2pt}\item
    \begin{tabular*}{0.97\textwidth}[t]{l@{\extracolsep{\fill}}r}
      \textbf{#1} & #2 \\
      \textit{\small #3} & \textit{\small #4} \\
    \end{tabular*}\vspace{-7pt}
}

% Publication line: bold title (wraps if long) ............... venue, flush right
\newcommand{\resumePub}[2]{%
  \item
    \begin{tabularx}{0.97\textwidth}[t]{@{}>{\raggedright\arraybackslash}X@{\hspace{1.5em}}r@{}}
      \small\textbf{#1} & #2 \\
    \end{tabularx}\vspace{-2pt}
}
